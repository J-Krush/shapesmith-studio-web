    **Step C — `netlify/functions/submit-quote/submit-quote.js` (NEW).** CommonJS handler:
    ```js
    const { Resend } = require('resend');

    const resend = new Resend(process.env.RESEND_API_KEY);
    const RECAPTCHA_THRESHOLD = 0.5;
    const MAX_BODY_BYTES = 100 * 1024; // 100KB — server-side memory bound (QTE-09)
    const FETCH_TIMEOUT_MS = 5000;

    const fetchWithTimeout = async (url, options) => {
      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);
      try {
        return await fetch(url, { ...options, signal: controller.signal });
      } finally {
        clearTimeout(timer);
      }
    };

    const formatUsd = (n) => {
      if (n >= 1000) return `$${(n / 1000).toFixed(1)}k`;
      return `$${Math.round(n)}`;
    };

    const formatQuoteText = (p) => {
      const m = p.metadata || {};
      const range = `${formatUsd(m.priceLow)} – ${formatUsd(m.priceHigh)}`; // U+2013, never a point
      const lines = [
        `Service: ${p.service}`,
        `From: ${p.name} <${p.email}>`,
        '',
        `File: ${m.filename} (${m.fileSizeMB} MB)`,
      ];
      if (m.volumeCm3 != null) lines.push(`Volume: ${m.volumeCm3} cm³`);
      if (m.bbox) lines.push(`Bounding box: ${m.bbox}`);
      if (m.triangleCount != null) lines.push(`Triangles: ${m.triangleCount}`);
      if (m.cutArea) lines.push(`Cut area: ${m.cutArea}`);
      if (m.pathLengthMm != null) lines.push(`Total cut length: ${m.pathLengthMm} mm`);
      lines.push(`Material: ${m.materialName}`);
      lines.push(`Quantity: ${m.quantity}`);
      lines.push(`Estimated range: ${range}`);
      if (p.message) lines.push('', `Message: ${p.message}`);
      lines.push('', 'Reply to this email with a real quote.');
      return lines.join('\n');
    };

    exports.handler = async (event) => {
      if (event.httpMethod !== 'POST') {
        return { statusCode: 405, body: 'Method not allowed' };
      }

      // Server-side memory bound (QTE-09) — reject oversized payloads before parsing.
      if (event.body && event.body.length > MAX_BODY_BYTES) {
        return { statusCode: 400, body: 'Payload too large' };
      }

      let payload;
      try { payload = JSON.parse(event.body); }
      catch { return { statusCode: 400, body: 'Invalid JSON' }; }

      const required = ['name', 'email', 'service', 'recaptchaToken', 'metadata'];
      for (const k of required) {
        if (!payload[k]) return { statusCode: 400, body: `Missing ${k}` };
      }

      // Cheap email shape check — keeps obviously bogus payloads off Resend.
      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(payload.email)) {
        return { statusCode: 400, body: 'Invalid email' };
      }

      // reCAPTCHA v3 server verify.
      let verifyJson;
      try {
        const verifyRes = await fetchWithTimeout('https://www.google.com/recaptcha/api/siteverify', {
          method: 'POST',
          headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
          body: new URLSearchParams({
            secret: process.env.RECAPTCHA_SECRET_KEY,
            response: payload.recaptchaToken,
          }),
        });
        verifyJson = await verifyRes.json();
      } catch (e) {
        console.error('reCAPTCHA verify failed:', e);
        return { statusCode: 502, body: 'Spam check unavailable' };
      }
      if (!verifyJson.success || (verifyJson.score ?? 0) < RECAPTCHA_THRESHOLD) {
        return { statusCode: 403, body: 'Spam check failed' };
      }

      // Send via Resend (with timeout — Pitfall 10).
      try {
        await resend.emails.send({
          from: 'Shapesmith Studio <quotes@shapesmith.studio>', // verified sender domain required (Pitfall 8)
          to: ['jkrush@shapesmith.studio'],
          reply_to: payload.email,
          subject: `[Quote] ${payload.service} — ${payload.name}`,
          text: formatQuoteText(payload),
        });
      } catch (e) {
        console.error('Resend send failed:', e);
        return { statusCode: 502, body: 'Email send failed' };
      }

      return {
        statusCode: 200,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ ok: true }),
      };
    };
    ```

            inputLabel="Your name"
            labelFor="quote-name"
            inputType="text"
            inputId="quote-name"
            inputName="quote-name"

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Visitor browser → Netlify Function | First server-side endpoint in this project. Untrusted POST body crosses here. CORS allowlist defaults to same-origin (Netlify default for `/.netlify/functions/*`); no manual CORS config needed. |
| Netlify Function → Google reCAPTCHA siteverify | Outbound HTTPS with secret key as form-encoded parameter. Standard pattern. |
| Netlify Function → Resend API | Outbound HTTPS with `RESEND_API_KEY` as bearer auth. Standard pattern. |
| `RESEND_API_KEY`, `RECAPTCHA_SECRET_KEY` env vars | Server-only — must NEVER be inlined into the SPA bundle. Netlify Functions runtime exposes them via `process.env.*`; the SPA build (`react-scripts build`) only inlines `REACT_APP_*` vars (CRA contract). |
| `REACT_APP_RECAPTCHA_SITE_KEY` env var | Public — INTENDED to be inlined into the SPA bundle and visible in DevTools. The site key alone cannot mint tokens server-side; only the secret key can verify. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-03-01 | Spoofing | Forged email metadata | mitigate | The Function controls the `from` address (verified Resend domain) and `to` (hardcoded owner inbox). The visitor only controls `name`, `email`, `message`, `service`, and `metadata.*`. The visitor's email goes into `reply_to` only — does NOT change envelope-from. SPF/DKIM signing happens via Resend on the verified domain. |
| T-03-03-02 | Tampering | Client-tampered metadata (priceLow / priceHigh) | accept | The visitor can use DevTools to send any priceLow/priceHigh in the JSON body — the Function does NOT recalculate price server-side. **This is acceptable** because: (a) QTE-05 legal-posture lock — the email is informational, the owner manually replies with a real quote; (b) re-running the calculation server-side would require re-fetching pricing-rule from Sanity per submission — adds latency and a Sanity-down failure mode for negligible gain. The owner should treat the range in the email as advisory and verify with their own pricing reference. Documented in 03-03-SUMMARY.md. |
| T-03-03-03 | Repudiation | Spam / fake submissions | mitigate | reCAPTCHA v3 server-verified score ≥ 0.5 (RECAPTCHA_THRESHOLD constant; tunable post-launch per Pitfall 9). Score < 0.5 returns 403 BEFORE any Resend call — keeps Resend usage off bot traffic and below the 100/day free-tier ceiling. Plus: server-side payload validation (required fields, email regex, payload size cap) keeps malformed bodies off Resend. |
| T-03-03-04 | Information Disclosure | Resend API key in client bundle | mitigate | Anti-pattern guard via grep gate `! grep -rE "from 'resend'\\|require\\('resend'\\)" src/`. `resend` only in `netlify/functions/submit-quote/package.json`. The function-local package.json is invisible to webpack — Netlify build installs deps into the function directory only. |
| T-03-03-05 | Information Disclosure | reCAPTCHA secret in client bundle | mitigate | `RECAPTCHA_SECRET_KEY` only referenced via `process.env.*` in the Function file. Verified by grep `! grep -rE "RECAPTCHA_SECRET_KEY" src/`. The site key (public) goes through `REACT_APP_RECAPTCHA_SITE_KEY` — explicitly browser-safe per Google docs. |
| T-03-03-06 | Denial of Service | Resend rate-limit abuse | mitigate | reCAPTCHA gate filters bots at the front. Free-tier ceiling is 100/day — even if every visitor is real, studio expects << 100/day. If volume grows, raise the threshold to 0.7 (Pitfall 9 follow-up) and/or move to Resend paid tier. |
| T-03-03-07 | Denial of Service | Function timeout on slow Resend / reCAPTCHA | mitigate | 5-second `fetchWithTimeout` on both upstream calls. Netlify sync function timeout is 10s; with two 5s budgets in series we have headroom. If a path becomes hot, migrate to Background Functions per Pitfall 10. |
| T-03-03-08 | Denial of Service | Oversized request body | mitigate | `MAX_BODY_BYTES = 100 * 1024` (100KB) checked against `event.body.length` BEFORE `JSON.parse`. Returns 400 immediately. The actual payload is metadata only (no file contents) and should never exceed ~2KB; 100KB is a 50× safety margin. (QTE-09 server-side memory bound.) |
| T-03-03-09 | Tampering / XSS | Email body contains visitor-supplied strings | mitigate | Email is plain text (`text:` field, NOT `html:`). Visitor strings (`name`, `email`, `message`, `materialName`, `filename`) are interpolated into the plain-text body verbatim — there's no HTML rendering surface to inject into. Resend's plain-text path does no HTML interpretation. |
| T-03-03-10 | Tampering / Email injection | Visitor-supplied newlines in `name` or `email` fields creating fake headers | mitigate | The Resend SDK builds the headers from the typed `from`/`to`/`subject` properties — visitor input only enters `subject` (sanitize-able) and `text` body. The simple email regex `/^[^\s@]+@[^\s@]+\.[^\s@]+$/` rejects any email with whitespace including `\r\n`. The `subject` line uses `payload.service` (an enum value) and `payload.name` (free-form); a visitor stuffing `\r\nBcc: attacker@example.com` into `name` would land it inside the `subject` string — Resend treats subjects as a single string and folds long lines per RFC 5322. **Defense:** add `payload.name = payload.name.replace(/[\r\n]/g, ' ')` before building the subject line. Add this normalization to the Function as a one-line guard. |
| T-03-03-11 | Elevation of Privilege | DOMPurify deferred (D-22) | accept | Plan 03-03 does NOT render user SVG to the DOM. The offscreen-host technique from Plan 03-01's `parseSvg.js` only reads geometry; it never appends to a visible container.
