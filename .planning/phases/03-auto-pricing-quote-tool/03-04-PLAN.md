    **Step A — `src/hooks/useLocalStorageState.jsx`.** Per RESEARCH Pattern 7. File extension is `.jsx` per CONVENTIONS.md (matches `useSanityQuery.jsx`).
    ```jsx
    import { useState, useEffect } from 'react';

    const useLocalStorageState = (key, initialValue) => {
      const [value, setValue] = useState(() => {
        try {
          const stored = window.localStorage.getItem(key);
          return stored !== null ? JSON.parse(stored) : initialValue;
        } catch {
          return initialValue;
        }
      });

      useEffect(() => {
        try {
          if (value === undefined || value === null) {
            window.localStorage.removeItem(key);
          } else {
            window.localStorage.setItem(key, JSON.stringify(value));
          }
        } catch {
          // QuotaExceededError or disabled storage — fail silently. UI still works in-memory.
        }
      }, [key, value]);

      return [value, setValue];
    };

    export default useLocalStorageState;
    ```

       import useLocalStorageState from '../../hooks/useLocalStorageState';
       import QuoteRestoreBanner from './QuoteRestoreBanner';

       const STORAGE_KEY = 'shapesmith-quote-v1';
       const DEFAULT_STATE = { tab: 'print', materialId: '', quantity: 1 };

       const QuoteTabs = () => {
         const [quoteState, setQuoteState] = useLocalStorageState(STORAGE_KEY, DEFAULT_STATE);
         // Restore-banner visibility: true when storage had a non-default value AT MOUNT.
         const [showRestoreBanner, setShowRestoreBanner] = useState(() => {
           const init = quoteState;
           return init && (init.materialId !== '' || init.quantity !== 1 || init.tab !== 'print');
         });

         const updateState = (patch) => {
           setQuoteState((prev) => ({ ...(prev ?? DEFAULT_STATE), ...patch }));
           setShowRestoreBanner(false); // first interaction dismisses the banner
         };

         const handleStartOver = () => {
           setQuoteState(null); // removes the key
           // After removal the hook still holds the old in-memory value; explicitly seed defaults next render via setQuoteState in a follow-up tick:
           setTimeout(() => setQuoteState(DEFAULT_STATE), 0);
           setShowRestoreBanner(false);
           // Also reset file/geometry/error per-tab state since the restore banner appears alongside file-already-dropped UX.
           setFilesByTab({ print: null, laser: null });
           setGeometryByTab({ print: null, laser: null });
           setErrorByTab({ print: '', laser: '' });

         // ...derive activeKey from quoteState.tab, materialId from quoteState.materialId, quantity from quoteState.quantity
         const activeKey = quoteState?.tab ?? 'print';
         const materialId = quoteState?.materialId ?? '';
         const quantity = quoteState?.quantity ?? 1;

       useEffect(() => {
         // If we restored from localStorage, don't override with the URL-based cascade.
         if (showRestoreBanner) return;
         const queryService = new URLSearchParams(window.location.search).get('service');
         if (queryService) {
           const m = SERVICES.find((s) => s.key === queryService || s.urlSegment === queryService);
           if (m) { updateState({ tab: m.key }); return; }

         const referrer = (typeof document !== 'undefined' && document.referrer) || '';
         const rule = SERVICES.find((s) => referrer.includes(`/${s.urlSegment}`));
         if (rule) updateState({ tab: rule.key });
