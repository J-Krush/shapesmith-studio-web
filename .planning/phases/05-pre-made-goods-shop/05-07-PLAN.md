```js
// netlify/functions/snipcart-order-webhook/formatOrderEmail.js
//
// Pure function — unit-testable. Builds a plain-text owner email body from
// the Snipcart order.completed payload. Mirrors the idiom used by
// netlify/functions/submit-quote/formatQuoteText.js (CommonJS export, line
// arrays joined with '\n', defensive optional-chain reads).
//
// Source: RESEARCH §"Pattern 5: Webhook Signature Validation" — formatOrderEmail.js code block.

const formatOrderEmail = (order) => {
  if (!order) return 'Order data missing — see Snipcart dashboard.';

  const lines = [];
  lines.push(`New order: ${order.invoiceNumber ?? '???'}`);
  const itemsCount = order.itemsCount ?? 0;
  lines.push(
    `Total: $${order.finalGrandTotal ?? '?'} (${itemsCount} item${itemsCount === 1 ? '' : 's'})`
  );
  lines.push('');
  lines.push('Customer:');
  lines.push(`  ${order.billingAddress?.fullName ?? 'unknown'}`);
  lines.push(`  ${order.email ?? 'no email'}`);
  if (order.phone) lines.push(`  ${order.phone}`);
  lines.push('');
  lines.push('Ship to:');
  const ship = order.shippingAddress ?? order.billingAddress ?? {};
  lines.push(`  ${ship.fullName ?? ''}`);
  lines.push(`  ${ship.address1 ?? ''}${ship.address2 ? ` ${ship.address2}` : ''}`);
  lines.push(`  ${ship.city ?? ''}, ${ship.province ?? ''} ${ship.postalCode ?? ''}`);
  lines.push(`  ${ship.country ?? ''}`);
  lines.push('');
  lines.push(`Shipping method: ${order.shippingMethod ?? 'n/a'} ($${order.shippingFees ?? 0})`);
  lines.push('');
  lines.push('Items:');
  for (const item of order.items ?? []) {
    lines.push(`  - ${item.quantity}× ${item.name} — $${item.totalPrice}`);
    if (item.description) lines.push(`      ${item.description}`);
  }
  lines.push('');
  if (order.token) {
    lines.push(`Dashboard: https://app.snipcart.com/dashboard/orders/${order.token}`);
  } else {
    lines.push('Dashboard: https://app.snipcart.com/dashboard/orders/');
  }
  return lines.join('\n');
};

module.exports = formatOrderEmail;
```

```js
// netlify/functions/snipcart-order-webhook/snipcart-order-webhook.js
//
// Snipcart order.completed webhook handler (CONTEXT D-15).
// Validates Snipcart's request token via app.snipcart.com/api/requestvalidation/{token}
// (RESEARCH §Pattern 5 — token-callback validation, NOT HMAC).
// Builds a custom-formatted owner email and sends via Resend (Phase 3 precedent).
// Returns 200 on Resend failure to suppress Snipcart retry storm (RESEARCH Pitfall 3).
//
// Source: RESEARCH §"Pattern 5: Webhook Signature Validation".

const { Resend } = require('resend');
const formatOrderEmail = require('./formatOrderEmail');

const resend = new Resend(process.env.RESEND_API_KEY);
const FETCH_TIMEOUT_MS = 5000;
const MAX_BODY_BYTES = 50 * 1024; // 50KB ceiling for an order payload

const fetchWithTimeout = async (url, options = {}) => {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);
  try {
    return await fetch(url, { ...options, signal: controller.signal });
  } finally {
    clearTimeout(timer);
  }
};

exports.handler = async (event) => {
  if (event.httpMethod !== 'POST') {
    return { statusCode: 405, body: 'Method not allowed' };
  }

  if (event.body && event.body.length > MAX_BODY_BYTES) {
    return { statusCode: 400, body: 'Payload too large' };
  }

  // 1. Validate Snipcart token (header is lowercased by Netlify).
  const token =
    event.headers['x-snipcart-requesttoken'] ||
    event.headers['X-Snipcart-RequestToken'];
  if (!token) {
    return { statusCode: 401, body: 'Missing webhook token' };
  }
  try {
    const verifyRes = await fetchWithTimeout(
      `https://app.snipcart.com/api/requestvalidation/${encodeURIComponent(token)}`
    );
    if (!verifyRes.ok) {
      console.error('Snipcart token validation rejected:', verifyRes.status);
      return { statusCode: 401, body: 'Invalid webhook token' };
    }
  } catch (e) {
    console.error('Snipcart token validation error:', e);
    // Fail-closed: forged payloads must not slip through during a Snipcart outage.
    // Snipcart retries non-2xx for ~24h, so a 30s outage replays once recovered.
    return { statusCode: 502, body: 'Webhook validation unavailable' };
  }

  // 2. Parse + filter event types.
  let payload;
  try {
    payload = JSON.parse(event.body);
  } catch {
    return { statusCode: 400, body: 'Invalid JSON' };
  }
  if (payload.eventName !== 'order.completed') {
    // Acknowledge other event types (the dashboard subscription should restrict
    // to order.completed only; this is defensive against misconfiguration).
    return { statusCode: 200, body: 'OK (ignored event)' };
  }

  // 3. Send Resend email to owner with full order info.
  try {
    await Promise.race([
      resend.emails.send({
        from: 'Shapesmith Studio <orders@shapesmith.studio>',
        to: ['jkrush@shapesmith.studio'],
        reply_to: payload.content?.email ?? 'noreply@shapesmith.studio',
        subject: `[Order ${payload.content?.invoiceNumber ?? '???'}] ${payload.content?.itemsCount ?? '?'} item(s) — $${payload.content?.finalGrandTotal ?? '?'}`,
        text: formatOrderEmail(payload.content),
      }),
      new Promise((_, reject) =>
        setTimeout(() => reject(new Error('Resend timeout')), FETCH_TIMEOUT_MS)
      ),
    ]);
  } catch (e) {
    // Critical deviation from submit-quote.js: this Function returns 200 even
    // when Resend fails. Snipcart retries on non-2xx responses, which would
    // cascade into an email-storm if Resend is intermittently flaky. The order
    // is still recorded in Snipcart's dashboard, and the owner gets Snipcart's
    // default email (CONTEXT D-15 keeps both). Single missed custom email is
    // an acceptable tradeoff vs. retry-storm risk.
    // (PATTERNS.md "Cross-Cutting Deviations" item 9 documents this.)
    console.error('Resend send failed (non-fatal):', e);
    console.error('Order received but custom email failed — see Snipcart dashboard:', payload.content?.invoiceNumber);
    return { statusCode: 200, body: 'OK (email send failed; logged)' };
  }

  return {
    statusCode: 200,
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ ok: true }),
  };
};
```

**Hard rules:**
- DO NOT switch token validation to HMAC — Snipcart explicitly uses token-callback (RESEARCH key-finding 4). There is no shared secret to HMAC against.
- DO NOT use `process.env.SNIPCART_API_SECRET` — the validation endpoint requires NO API key (RESEARCH assumption A3 + Security Domain row "Forged webhook POST"). If A3 turns out wrong during Plan 08 UAT, the Function will return 401s and the executor will know to add the secret. NOT in this plan.
- DO NOT return 4xx/5xx on Resend failure — return 200 (deviation 9 explicitly documented in PATTERNS.md). Snipcart retries non-2xx responses; cascading retries multiply errors.
- DO NOT skip the `event.eventName !== 'order.completed'` guard — if the dashboard is misconfigured to send all events, we'd email the owner for every event type.
- DO NOT trust `payload.content` to exist — use optional chaining for every field access (the formatter handles missing fields too via Task 1 defensive coding).
- DO NOT log full PII (full address blocks, phone numbers) — `console.error` only the invoice number for debugging when Resend fails. Customer email and address are NOT logged.
