**Step 2 — `netlify/functions/snipcart-validate-product/snipcart-validate-product.js`**:

```js
// netlify/functions/snipcart-validate-product/snipcart-validate-product.js
//
// Snipcart JSON crawler endpoint. Called by Snipcart's order-validation server
// BEFORE confirming any order. Snipcart compares the response JSON against the
// data-item-* attributes that came from the cart; if they don't match, the order
// is rejected. This is the SPA-safe alternative to the HTML crawler — without
// it, Snipcart's HTML crawler would hit /shop/{slug} and get back the SPA's
// index.html with no embedded product data, silently failing validation.
//
// Set Content-Type: application/json so Snipcart routes to the JSON validator
// (NOT the HTML crawler).
//
// Required JSON fields per docs.snipcart.com/v3/setup/order-validation:
//   id    — must equal data-item-id from the buy button
//   price — must equal data-item-price
//   url   — must equal data-item-url
// (name + image optional but recommended for parity)
//
// This Function does NOT require auth — it's intentionally public-readable;
// Sanity's anonymous CDN access is the read-side trust model.
//
// Source: RESEARCH §"Pattern 4: JSON Crawler Validation Endpoint (Required for SPA)".

const SANITY_PROJECT = 'qx9kep1e';
const SANITY_DATASET = 'production';
const API_VERSION = '2023-06-16';

const SLUG_REGEX = /^[a-z0-9-]+$/i;
const SLUG_MAX_LEN = 200;
const FETCH_TIMEOUT_MS = 5000;

const fetchWithTimeout = async (url, options = {}) => {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);
  try {
    return await fetch(url, { ...options, signal: controller.signal });
  } finally {
    clearTimeout(timer);
  }
};

exports.handler = async (event) => {
  if (event.httpMethod !== 'GET') {
    return { statusCode: 405, body: 'Method not allowed' };
  }

  const slug = event.queryStringParameters?.slug;
  if (!slug || slug.length > SLUG_MAX_LEN || !SLUG_REGEX.test(slug)) {
    return { statusCode: 400, body: 'Missing or invalid slug' };
  }

  // Build a parameterized GROQ query. The query string is fixed; the slug binds
  // to the $slug parameter via the URL — no string concatenation, no GROQ injection.
  const groq = encodeURIComponent(
    `*[_type=="product" && slug.current==$slug][0]{
      "id": slug.current,
      name,
      price,
      description,
      stockQuantity,
      "image": images[0].asset->url
    }`
  );
  const slugParam = encodeURIComponent(JSON.stringify(slug)); // JSON-encode then URI-encode
  const sanityUrl =
    `https://${SANITY_PROJECT}.apicdn.sanity.io/v${API_VERSION}/data/query/${SANITY_DATASET}` +
    `?query=${groq}&%24slug=${slugParam}`;

  let result;
  try {
    const res = await fetchWithTimeout(sanityUrl);
    if (!res.ok) {
      console.error('Sanity fetch non-OK:', res.status);
      return { statusCode: 502, body: 'Sanity fetch failed' };
    }
    const json = await res.json();
    result = json.result;
  } catch (e) {
    console.error('Sanity fetch error:', e);
    return { statusCode: 502, body: 'Sanity unreachable' };
  }

  if (!result) {
    // Product was unpublished after the customer added it to cart — refuse the order.
    return { statusCode: 404, body: 'Product not found' };
  }

  // Reconstruct the URL the buy button sent. Snipcart will compare its request
  // URL to this `url` field and reject the order if they differ. We must echo
  // back exactly what AddToCartButton.jsx (Plan 05) emits.
  const baseUrl =
    process.env.URL ||
    (event.headers.host ? `https://${event.headers.host}` : 'https://shapesmith.studio');
  const productUrl = `${baseUrl}/.netlify/functions/snipcart-validate-product?slug=${slug}`;

  const responseBody = {
    id: result.id,
    name: result.name,
    price: Number(result.price),
    url: productUrl,
    description: result.description ?? '',
    // Stock guard — defensive secondary check for inventory race (RESEARCH Pitfall 5).
    // Primary guard is data-item-max-quantity on the button.
    stock: result.stockQuantity,
  };
  if (result.image) {
    responseBody.image = result.image;
  }

  return {
    statusCode: 200,
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(responseBody),
  };
};
```

**Hard rules:**
- DO NOT use `@sanity/client` in this Function. Raw HTTPS to apicdn.sanity.io keeps the deployed Function bundle small (esbuild bundles all deps; @sanity/client is ~50KB minified).
- DO NOT skip the slug regex validation — it's the GROQ injection mitigation (RESEARCH §Security Domain).
- DO NOT change `Content-Type: application/json` — that header is the trigger for Snipcart's JSON validator (RESEARCH Pattern 4 critical note).
- DO NOT log the full Sanity response body or the slug query parameter beyond `console.error` for failures (PII isn't an issue here, but log hygiene is). Log the HTTP status code only.
- DO NOT add `data-item-max-quantity` to the response — that's a button-side attribute, not a JSON-crawler field. The `stock` field IS what we communicate at validation time.
- DO NOT include `stackable`, `taxable`, `shippable`, `weight`, or any other data-item-* in the response body — Snipcart only validates `id`, `price`, `url` as required (and softly checks `name`, `description`, `image` for parity); we don't need to reproduce the full attribute surface.
